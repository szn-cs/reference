			  #!/usr/bin/env bash
			  
			  # Exit immediately if a command exits with a non-zero status
			  set -e
			  
			  # ANSI Color codes for clean interface logging
			  RED='\033[0;31m'
			  GREEN='\033[0;32m'
			  YELLOW='\033[0;33m'
			  BLUE='\033[0;34m'
			  NC='\033[0m' # No Color
			  
			  echo -e "${BLUE}=================================================================${NC}"
			  echo -e "${BLUE} Starting Modern Fedora 44 + Technitium DNS Pipeline Setup     ${NC}"
			  echo -e "${BLUE}=================================================================${NC}"
			  
			  # 1. Root Privilege Escalation Check
			  if [ "$EUID" -ne 0 ]; then
			    echo -e "${RED}❌ Error: This script modifies system networking loops. Run with sudo.${NC}" >&2
			    exit 1
			  fi
			  
			  # 2. Automated Target Profile Identification
			  echo -e "${YELLOW}[Step 1/4] Automatically identifying active NetworkManager profile...${NC}"
			  TARGET_PROFILE=$(nmcli -g NAME,DEVICE connection show --active | grep 'enp4s0' | cut -d: -f1 || true)
			  
			  if [ -z "$TARGET_PROFILE" ]; then
			    echo -e "${RED}❌ Error: No active NetworkManager connection profile found for interface enp4s0.${NC}" >&2
			    exit 1
			  fi
			  echo -e "${GREEN}Found Active Profile: \"$TARGET_PROFILE\"${NC}"
			  
			  # 3. Restrict NetworkManager Profile Leases
			  echo -e "\n${YELLOW}[Step 2/4] Stripping hardware interface DHCP DNS tables...${NC}"
			  nmcli connection modify "$TARGET_PROFILE" ipv4.ignore-auto-dns yes
			  nmcli connection modify "$TARGET_PROFILE" ipv6.ignore-auto-dns yes
			  nmcli connection modify "$TARGET_PROFILE" ipv4.dns ""
			  nmcli connection modify "$TARGET_PROFILE" ipv6.dns ""
			  
			  echo -e "${BLUE}🔄 Bouncing NetworkManager connection profile: $TARGET_PROFILE...${NC}"
			  nmcli connection down "$TARGET_PROFILE" && nmcli connection up "$TARGET_PROFILE"
			  
			  # 4. Flush Active Link Hardware Scopes
			  echo -e "\n${YELLOW}[Step 3/4] Dropping dynamic fallback scopes on enp4s0...${NC}"
			  if ip link show enp4s0 &>/dev/null; then
			    resolvectl revert enp4s0 || true
			  fi
			  
			  # 5. Establish Modern Direct Interface Pointer
			  echo -e "\n${YELLOW}[Step 4/4] Structuring direct loopback address configuration...${NC}"
			  rm -f /etc/resolv.conf
			  echo "nameserver 127.0.0.1" > /etc/resolv.conf
			  
			  # 6. Reload System Control Daemons
			  echo -e "\n${YELLOW}Re-indexing system configs and clearing RAM caches...${NC}"
			  systemctl daemon-reload
			  systemctl restart systemd-resolved
			  resolvectl flush-caches || true
			  
			  # =================================================================
			  # AUTOMATED VERIFICATION PHASE
			  # =================================================================
			  echo -e "\n${BLUE}=================================================================${NC}"
			  echo -e "${BLUE} Running Post-Configuration Verification Suite                 ${NC}"
			  echo -e "${BLUE}=================================================================${NC}"
			  
			  # Allow the system sockets 1 second to bind completely before testing
			  sleep 1
			  
			  # Verification Test A: Interface Scope Leaks
			  echo -ne "🔍 Verification A: Checking link-level scopes... "
			  LINK_SCOPE_COUNT=$(resolvectl status enp4s0 2>/dev/null | grep -i "Current Scopes:" | awk '{print $3}' || echo "none")
			  
			  if [ "$LINK_SCOPE_COUNT" = "none" ] || [ -z "$LINK_SCOPE_COUNT" ]; then
			    echo -e "${GREEN}PASS (enp4s0 is completely unmanaged)${NC}"
			  else
			    echo -e "${RED}FAIL (enp4s0 has active scope: $LINK_SCOPE_COUNT)${NC}"
			  fi
			  
			  # Verification Test B: Direct Resolve Resolution Architecture
			  echo -ne "🔍 Verification B: Testing system application query path... "
			  
			  # Perform an nslookup and parse out the active resolving server address
			  RESOLVER_IP=$(nslookup google.com 2>/dev/null | grep -i "Server:" | awk '{print $2}' || echo "FAILED")
			  
			  if [ "$RESOLVER_IP" = "127.0.0.1" ]; then
			    echo -e "${GREEN}PASS (Traffic securely locked to Technitium on 127.0.0.1)${NC}"
			  else
			    echo -e "${RED}FAIL (Traffic routed outside target space via: $RESOLVER_IP)${NC}"
			  fi
			  
			  echo -e "\n${BLUE}=================================================================${NC}"
			  if [ "$RESOLVER_IP" = "127.0.0.1" ]; then
			    echo -e "${GREEN} 🎉 SYSTEM FULLY OPTIMIZED AND VERIFIED SUCCESFULLY!${NC}"
			  else
			    echo -e "${RED} ❌ PIPELINE MISALIGNED. Verify Technitium service status.${NC}"
			  fi
			  echo -e "${BLUE}=================================================================${NC}"
			  ```
