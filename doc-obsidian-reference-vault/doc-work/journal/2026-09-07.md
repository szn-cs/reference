# Fedora network advertising

#Linux
- obfuscate device name `hide-device-name.sh`
    - ```
      #!/usr/bin/env bash
      # Stop Fedora from disclosing the device name on the network.
      #
      #   sudo bash hide-device-name.sh          # apply (default; idempotent)
      #   sudo bash hide-device-name.sh revert   # reverse everything this script did
      set -euo pipefail
      CONF_FILE=/etc/NetworkManager/conf.d/01-no-name-advertising.conf
      ZONE=FedoraWorkstation
      MARKER="# managed by hide-device-name.sh"
      reapply_devices() {
          for dev in $(nmcli -t -f DEVICE,STATE device status 2>/dev/null | awk -F: '$2=="connected"{print $1}'); do
              nmcli device reapply "$dev" 2>/dev/null || true
          done
      }
      reload_nm() {
          systemctl reload NetworkManager 2>/dev/null || systemctl restart NetworkManager
      }
      apply() {
          # 1. Global NetworkManager defaults: every current and FUTURE connection
          #    profile inherits these unless it overrides them explicitly.
          #    - mdns/llmnr off: systemd-resolved stops answering <hostname>.local /
          #      NetBIOS-style name queries from other LAN devices
          #    - dhcp-send-hostname off: the router/hotspot no longer learns a name
          #      for this machine via DHCP
          cat > "$CONF_FILE" <<EOF
      $MARKER
      [connection]
      connection.mdns=0
      connection.llmnr=0
      ipv4.dhcp-send-hostname=false
      ipv6.dhcp-send-hostname=false
      EOF
          # Pick up the new defaults and push them onto the live devices
          reload_nm
          reapply_devices
          # 2. Defense in depth: stop answering inbound mDNS at the firewall.
          #    (Outbound .local lookups for other devices keep working.)
          if firewall-cmd --permanent --zone="$ZONE" --query-service=mdns >/dev/null 2>&1; then
              firewall-cmd --permanent --zone="$ZONE" --remove-service=mdns
              firewall-cmd --reload
          else
              echo "mdns already removed from $ZONE zone"
          fi
          # 3. Verify
          echo "--- firewall services now:"
          firewall-cmd --zone="$ZONE" --list-services
          echo "--- global defaults now in effect ($CONF_FILE):"
          cat "$CONF_FILE"
          echo "--- per-link responders should all be negative (-mDNS -LLMNR):"
          resolvectl status wlp6s0 | grep Protocols || true
          resolvectl status enp4s0 | grep Protocols || true
          echo "Done."
      }
      revert() {
          # 1. Remove the global defaults file (only if this script owns it)
          if [[ -f "$CONF_FILE" ]]; then
              if grep -qF "$MARKER" "$CONF_FILE"; then
                  rm -f "$CONF_FILE"
                  echo "Removed $CONF_FILE"
              else
                  echo "WARNING: $CONF_FILE was modified outside this script; leaving it in place."
              fi
          else
              echo "No $CONF_FILE to remove"
          fi
          # 2. Restore mdns to the firewall zone
          if firewall-cmd --permanent --zone="$ZONE" --query-service=mdns >/dev/null 2>&1; then
              echo "mdns already present in $ZONE zone"
          else
              firewall-cmd --permanent --zone="$ZONE" --add-service=mdns
              firewall-cmd --reload
          fi
          # 3. Make NetworkManager and the live devices pick the original defaults up
          reload_nm
          reapply_devices
          # 4. Verify
          echo "--- firewall services now:"
          firewall-cmd --zone="$ZONE" --list-services
          echo "--- per-link responders should be positive again (+mDNS LLMNR=resolve):"
          resolvectl status wlp6s0 | grep Protocols || true
          resolvectl status enp4s0 | grep Protocols || true
          echo "Reverted."
      }
      case "${1:-apply}" in
          apply) apply ;;
          revert|undo) revert ;;
          *)
              echo "usage: sudo bash ${0##*/} [apply|revert]" >&2
              exit 1
              ;;
      esac
      ```
